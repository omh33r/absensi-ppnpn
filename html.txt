        // Haversine Formula: Calculate Distance between two Lat/Lng points in meters
        function calculateDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3; // metres
            const φ1 = lat1 * Math.PI/180;
            const φ2 = lat2 * Math.PI/180;
            const Δφ = (lat2-lat1) * Math.PI/180;
            const Δλ = (lon2-lon1) * Math.PI/180;

            const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                    Math.cos(φ1) * Math.cos(φ2) *
                    Math.sin(Δλ/2) * Math.sin(Δλ/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

            return Math.round(R * c); // Distance in meters
        }

        function initUserPositionWatcher() {
            if (navigator.geolocation) {
                navigator.geolocation.watchPosition(pos => {
                    userCurrentPos = {
                        lat: pos.coords.latitude,
                        lng: pos.coords.longitude
                    };
                    updateLocationStatus();
                }, err => {
                    // Fallback to office location if denied
                    updateLocationStatus();
                });
            } else {
                updateLocationStatus();
            }
        }

        function updateLocationStatus() {
            // Distance calculated based on office coords + simulated offset
            const realDist = calculateDistance(userCurrentPos.lat, userCurrentPos.lng, settings.officeLat, settings.officeLng);
            // We use simulated distance to allow easy testing in browser
            const currentDist = simulatedDistOffset; 

            const badge = document.getElementById('radiusBadge');
            const distVal = document.getElementById('distVal');
            const maxRad = document.getElementById('maxRadiusVal');
            const btnMasuk = document.getElementById('btnAbsenMasuk');
            const btnPulang = document.getElementById('btnAbsenPulang');

            if (distVal) distVal.innerText = `${currentDist} m`;
            if (maxRad) maxRad.innerText = `${settings.radiusMeter} m`;

            const isWithinRadius = currentDist <= settings.radiusMeter;

            if (badge) {
                if (isWithinRadius) {
                    badge.className = "px-3 py-1 rounded-full text-xs font-semibold bg-emerald-100 text-emerald-700 flex items-center gap-1";
                    badge.innerHTML = `<i data-lucide="check-circle" class="w-3.5 h-3.5"></i> Dalam Radius (${currentDist}m)`;
                    if(btnMasuk) btnMasuk.disabled = false;
                    if(btnPulang) btnPulang.disabled = false;
                } else {
                    badge.className = "px-3 py-1 rounded-full text-xs font-semibold bg-red-100 text-red-700 flex items-center gap-1";
                    badge.innerHTML = `<i data-lucide="x-circle" class="w-3.5 h-3.5"></i> Di Luar Radius (${currentDist}m)`;
                    if(btnMasuk) btnMasuk.disabled = true;
                    if(btnPulang) btnPulang.disabled = true;
                }
                lucide.createIcons();
            }

            // Update Leaflet marker position
            if (empMarker && empMap) {
                const simLat = settings.officeLat + (simulatedDistOffset * 0.000009); // slight shift for visual demo
                const simLng = settings.officeLng;
                empMarker.setLatLng([simLat, simLng]);
            }
        }

        function processAttendance(type) {
            const currentDist = simulatedDistOffset;
            if (currentDist > settings.radiusMeter) {
                showToast('Absensi Ditolak! Anda berada di luar radius 50 meter kantor.', 'error');
                return;
            }

            // Capture Snapshot Image from Camera
            const video = document.getElementById('empCamVideo');
            const canvas = document.getElementById('empCapCanvas');
            const context = canvas.getContext('2d');
            canvas.width = 320;
            canvas.height = 240;
            context.drawImage(video, 0, 0, 320, 240);
            const photoSnap = canvas.toDataURL('image/jpeg');

            const now = new Date();
            const currentTimeStr = now.toTimeString().substring(0, 5); // HH:MM

            // Calculate Late status based on work start & tolerance
            let statusWaktu = 'Tepat Waktu';
            if (type === 'MASUK') {
                const [startH, startM] = settings.workStart.split(':').map(Number);
                const limitMinutes = (startH * 60) + startM + parseInt(settings.lateToleranceMin);
                const currentMinutes = (now.getHours() * 60) + now.getMinutes();

                if (currentMinutes > limitMinutes) {
                    statusWaktu = 'Terlambat';
                }
            }

            const record = {
                id: Date.now().toString(),
                userId: currentUser.id,
                userName: currentUser.name,
                type: type, // 'MASUK' or 'PULANG'
                timestamp: now.toISOString(),
                dateStr: now.toLocaleDateString('id-ID'),
                timeStr: now.toLocaleTimeString('id-ID'),
                distance: currentDist,
                statusWaktu: statusWaktu,
                lat: settings.officeLat + (simulatedDistOffset * 0.000009),
                lng: settings.officeLng,
                photo: photoSnap
            };

            attendanceLogs.unshift(record);
            localStorage.setItem('ppnpn_attendance', JSON.stringify(attendanceLogs));

            showToast(`Presensi ${type} Berhasil Ditambahkan!`, 'success');
            renderPersonalLogs();
        }

            `).join('');
        }

        function exportToCSV() {
            if (attendanceLogs.length === 0) {
                showToast('Tidak ada data untuk diekspor', 'error');
                return;
            }

            let csvContent = "data:text/csv;charset=utf-8,Tanggal,Waktu,Nama PPNPN,Tipe Absen,Status Waktu,Jarak (Meter)\n";
            attendanceLogs.forEach(row => {
                csvContent += `"${row.dateStr}","${row.timeStr}","${row.userName}","${row.type}","${row.statusWaktu}","${row.distance}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Rekap_Absensi_PPNPN_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
